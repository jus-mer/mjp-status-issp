ntile_na <- function(x,n)
{
  notna <- !is.na(x)
  out <- rep(NA_real_,length(x))
  out[notna] <- ntile(x[notna],n)

# ntile function without NAs
ntile_zero_na <- function(x, n = 10) {
  out   <- rep(NA_real_, length(x))
  is_na <- is.na(x)
  is0   <- !is_na & x == 0
  pos   <- !is_na & x > 0
  out[is0] <- 0
  if (any(pos)) out[pos] <- ntile(x[pos], n)
  out
}

# ntile function without NAs
ntile_zero_na <- function(x, n = 10) {
  out   <- rep(NA_real_, length(x))
  is_na <- is.na(x)
  is0   <- !is_na & x == 0
  pos   <- !is_na & x > 0
  out[is0] <- 0
  if (any(pos)) out[pos] <- ntile(x[pos], n)
  out
}
